FIX (1) parse_request body-framing (self/serve/http_server.hexa :24-29 -> :24-34) — OP-92 pattern.
UNFIXED:
    let parts = raw.split("\r\n\r\n")     // splits on EVERY blank line
    let header_block = parts[0]
    if len(parts) > 1 { body = parts[1] } // keeps ONLY first body segment -> TRUNCATION
FIXED (split on the FIRST "\r\n\r\n" only — RFC-7230 framing):
    let sep = raw.index_of("\r\n\r\n")
    let mut header_block = raw
    if sep >= 0 {
        header_block = raw.substring(0, sep)
        body = raw.substring(sep + 4, len(raw))   // whole remainder, CRLFs included
    }

FIX (2) parse_query value-split (self/serve/http_server.hexa :78-89) — OP-91 pattern.
UNFIXED:
    let kv = pairs[i].split("=")          // shatters on EVERY "="
    if len(kv) >= 2 { result[kv[0]] = kv[1] }        // value-trunc on "=" in value
    else { if len(kv) == 1 { result[kv[0]] = "" } }  // empty pair -> spurious "" key
FIXED (split on the FIRST "=" only + skip empty pairs):
    let pair = pairs[i]
    if len(pair) > 0 {
        let eq = pair.index_of("=")
        if eq >= 0 {
            result[pair.substring(0, eq)] = pair.substring(eq + 1, len(pair))
        } else {
            result[pair] = ""
        }
    }

// (runtime_cuda.c — cublasDgemmStridedBatched OP_T, strideA/strideB=0 broadcast

// DELEGATES per slice to hexa_farr_matmul through the identical host transpose.
// Never silently wrong, only silently slower.
//
// BYTE-EQ FIX (2026-06-02): the original no-CUDA host loop was an open-coded
// ijk dot (acc += A·B); against hexa_farr_matmul's ikj UNROLLED (FMA-contracted)
// reference it drifted ~1e-14 — BT-BATCHED max|Δ|=2.84e-14 — FAILING the

/* ═══════════════════════════════════════════════════════════════════
 * LEVER (3) — BATCHED transpose-aware GEMM-feed wrappers.
 *
 * Drop the DOMINANT 65% batched-expert host repack: feed the raw
 * weights w[Cout,Kdim] / grads dy[T,Cout] DIRECTLY through ONE
 * strided-batched op-flag GEMM, with a strideA/strideB=0 broadcast that
 * collapses the shared-xcol duplication. CUDA → the runtime_cuda.c
 * device kernel; no-CUDA → the byte-eq host loop (the
 * clm_batched_gemmfeed_eq.hexa reference).
 *
 * _bt : per slice g, C_g[M,N] = A_g[M,K] @ B_g^T (B_g stored [N,K]).
 *       as_blocks=0 → broadcast the shared A (the fwd xcol); 1 → per-prob.
 * _atb: per slice g, C_g[M,N] = A_g^T @ B_g (A_g stored [K,M] = dy).
 *       bs_blocks=0 → broadcast the shared B (the bwd xcol); 1 → per-prob.
 * ═══════════════════════════════════════════════════════════════════ */
HexaVal hexa_forge_dispatch_matmul_batched_bt(HexaVal a_v, HexaVal m_v, HexaVal k_v,
                                              HexaVal b_v, HexaVal n_v,
                                              HexaVal batch_v, HexaVal as_v,
                                              HexaVal c_v) {
    int64_t a_id  = hexa_as_num(a_v);
    int64_t M     = hexa_as_num(m_v);
    int64_t K     = hexa_as_num(k_v);
    int64_t b_id  = hexa_as_num(b_v);
    int64_t N     = hexa_as_num(n_v);
    int64_t batch = hexa_as_num(batch_v);
    int64_t as    = hexa_as_num(as_v);   /* a_stride_blocks ∈ {0,1} */
    int64_t c_id  = hexa_as_num(c_v);
    if (M <= 0 || K <= 0 || N <= 0 || batch <= 0) return hexa_int(-1);
    if (as != 0 && as != 1) return hexa_int(-1);
    if (a_id < 0 || a_id >= _hx_farr_count) return hexa_int(-1);
    if (b_id < 0 || b_id >= _hx_farr_count) return hexa_int(-1);
    if (c_id < 0 || c_id >= _hx_farr_count) return hexa_int(-1);
    if (_hx_farr_table[c_id].len < batch * M * N) return hexa_int(-1);
#ifdef HEXA_CUDA
    extern int _hx_cuda_farr_matmul_batched_bt_gpu(int64_t, int64_t, int64_t,
                                                   int64_t, int64_t, int64_t,
                                                   int64_t, int64_t);
    int grc = _hx_cuda_farr_matmul_batched_bt_gpu(a_id, M, K, b_id, N, batch, as, c_id);
    if (grc == 0) return hexa_int(0);
    /* GPU path failed — fall through to the host loop (correctness kept). */
#endif
    /* no-CUDA host fallback — BYTE-EQ contract (the HARD gate, max|Δ|=0.0).
     * The oracle reference is per-slice host-transpose w_g[N,K] → Wt[K,N] THEN
     * hexa_farr_matmul(Ag,M,K, Wt,N). An open-coded ijk dot drifts ~1e-14 from
     * hexa_farr_matmul's ikj unrolled (FMA-contracted) loop, so DELEGATE per
     * slice through the identical Wt host transpose — bit-identical by
     * construction. The farr id-array may realloc inside hexa_farr_zeros/
     * _matmul, so re-fetch buf ptrs by id every iteration. */
    {
        for (int64_t g = 0; g < batch; g++) {
            HexaVal wt_h = hexa_farr_zeros(hexa_int(K * N));
            int64_t wt_id = HX_INT(wt_h);
            if (wt_id < 0 || wt_id >= _hx_farr_count) return hexa_int(-1);
            const double* Bg = _hx_farr_table[b_id].buf + g * N * K;   /* w_g[N,K] */
            double* Wt = _hx_farr_table[wt_id].buf;
            for (int64_t n = 0; n < N; n++)
                for (int64_t kk = 0; kk < K; kk++)
                    Wt[kk * N + n] = Bg[n * K + kk];
            /* stage shared/own Ag[M,K] into a temp farr to pass its id. */
            HexaVal ag_h = hexa_farr_zeros(hexa_int(M * K));
            int64_t ag_id = HX_INT(ag_h);
            if (ag_id < 0 || ag_id >= _hx_farr_count) { hexa_farr_free(wt_h); return hexa_int(-1); }
            {
                const double* Asrc = _hx_farr_table[a_id].buf + (as == 0 ? 0 : g * M * K);
                double* Adst = _hx_farr_table[ag_id].buf;
                for (int64_t t = 0; t < M * K; t++) Adst[t] = Asrc[t];
            }
            HexaVal csl = hexa_farr_matmul(hexa_int(ag_id), hexa_int(M), hexa_int(K),
                                           hexa_int(wt_id), hexa_int(N));
            int64_t csl_id = HX_INT(csl);
            if (csl_id < 0 || csl_id >= _hx_farr_count) { hexa_farr_free(wt_h); hexa_farr_free(ag_h); return hexa_int(-1); }
            const double* Csrc = _hx_farr_table[csl_id].buf;
            double*       Cg   = _hx_farr_table[c_id].buf + g * M * N;
            for (int64_t t = 0; t < M * N; t++) Cg[t] = Csrc[t];
            hexa_farr_free(wt_h); hexa_farr_free(ag_h); hexa_farr_free(csl);
        }
    }
    return hexa_int(0);
}
HexaVal forge_dispatch_matmul_batched_bt(HexaVal a, HexaVal m, HexaVal k,
                                         HexaVal b, HexaVal n, HexaVal batch,
                                         HexaVal as, HexaVal c) {
    return hexa_forge_dispatch_matmul_batched_bt(a, m, k, b, n, batch, as, c);
}

HexaVal hexa_forge_dispatch_matmul_batched_atb(HexaVal a_v, HexaVal m_v, HexaVal k_v,
                                               HexaVal b_v, HexaVal n_v,
                                               HexaVal batch_v, HexaVal bs_v,
                                               HexaVal c_v) {
    int64_t a_id  = hexa_as_num(a_v);
    int64_t M     = hexa_as_num(m_v);
    int64_t K     = hexa_as_num(k_v);
    int64_t b_id  = hexa_as_num(b_v);
    int64_t N     = hexa_as_num(n_v);
    int64_t batch = hexa_as_num(batch_v);
    int64_t bs    = hexa_as_num(bs_v);   /* b_stride_blocks ∈ {0,1} */
    int64_t c_id  = hexa_as_num(c_v);
    if (M <= 0 || K <= 0 || N <= 0 || batch <= 0) return hexa_int(-1);
    if (bs != 0 && bs != 1) return hexa_int(-1);
    if (a_id < 0 || a_id >= _hx_farr_count) return hexa_int(-1);
    if (b_id < 0 || b_id >= _hx_farr_count) return hexa_int(-1);
    if (c_id < 0 || c_id >= _hx_farr_count) return hexa_int(-1);
    if (_hx_farr_table[c_id].len < batch * M * N) return hexa_int(-1);
#ifdef HEXA_CUDA
    extern int _hx_cuda_farr_matmul_batched_atb_gpu(int64_t, int64_t, int64_t,
                                                    int64_t, int64_t, int64_t,
                                                    int64_t, int64_t);
    int grc = _hx_cuda_farr_matmul_batched_atb_gpu(a_id, M, K, b_id, N, batch, bs, c_id);
    if (grc == 0) return hexa_int(0);
#endif
    /* no-CUDA host fallback — BYTE-EQ contract (max|Δ|=0.0). Reference is
     * per-slice host-transpose A_g[K,M] → AgT[M,K] THEN hexa_farr_matmul(AgT,
     * M,K, Bg,N) with Bg the shared/own B[K,N]. DELEGATE through the identical
     * transpose so the ikj accumulation matches bit-for-bit (open-coded ijk dot
     * would drift ~1e-14). Re-fetch buf ptrs by id after each alloc. */
    {
        for (int64_t g = 0; g < batch; g++) {
            HexaVal at_h = hexa_farr_zeros(hexa_int(M * K));
            int64_t at_id = HX_INT(at_h);
            if (at_id < 0 || at_id >= _hx_farr_count) return hexa_int(-1);
            const double* Ag = _hx_farr_table[a_id].buf + g * K * M;   /* A_g[K,M]=dy */
            double* AgT = _hx_farr_table[at_id].buf;
            for (int64_t kk = 0; kk < K; kk++)
                for (int64_t i = 0; i < M; i++)
                    AgT[i * K + kk] = Ag[kk * M + i];
            HexaVal bg_h = hexa_farr_zeros(hexa_int(K * N));
            int64_t bg_id = HX_INT(bg_h);
            if (bg_id < 0 || bg_id >= _hx_farr_count) { hexa_farr_free(at_h); return hexa_int(-1); }
            {
                const double* Bsrc = _hx_farr_table[b_id].buf + (bs == 0 ? 0 : g * K * N);
                double* Bdst = _hx_farr_table[bg_id].buf;
                for (int64_t t = 0; t < K * N; t++) Bdst[t] = Bsrc[t];
            }
            HexaVal csl = hexa_farr_matmul(hexa_int(at_id), hexa_int(M), hexa_int(K),
                                           hexa_int(bg_id), hexa_int(N));
            int64_t csl_id = HX_INT(csl);
            if (csl_id < 0 || csl_id >= _hx_farr_count) { hexa_farr_free(at_h); hexa_farr_free(bg_h); return hexa_int(-1); }
            const double* Csrc = _hx_farr_table[csl_id].buf;
            double*       Cg   = _hx_farr_table[c_id].buf + g * M * N;
            for (int64_t t = 0; t < M * N; t++) Cg[t] = Csrc[t];
            hexa_farr_free(at_h); hexa_farr_free(bg_h); hexa_farr_free(csl);
        }
    }
    return hexa_int(0);
}
HexaVal forge_dispatch_matmul_batched_atb(HexaVal a, HexaVal m, HexaVal k,
                                          HexaVal b, HexaVal n, HexaVal batch,
                                          HexaVal bs, HexaVal c) {
    return hexa_forge_dispatch_matmul_batched_atb(a, m, k, b, n, batch, bs, c);
}
